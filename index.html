function handleRateData(data) { if (data && data.rates && data.rates.IDR) { var rate = Math.round(data.rates.IDR); document.getElementById('exchange-rate').innerText = formatNumberCommas(rate); document.getElementById('rate-dot').className = 'status-dot live'; } } function fetchExchangeRate() { // 1. Try v4 endpoint via XHR legacyGet('http://api.exchangerate-api.com/v4/latest/AUD', function(data) { if (data && data.rates && data.rates.IDR) { handleRateData(data); } else { // 2. Try open.er-api v6 via XHR legacyGet('https://open.er-api.com/v6/latest/AUD', function(data2) { if (data2 && data2.rates && data2.rates.IDR) { handleRateData(data2); } else { // 3. Last resort: JSONP script tag var script = document.createElement('script'); script.src = 'https://open.er-api.com/v6/latest/AUD?callback=handleRateData'; document.body.appendChild(script); } }); } }); }
